# Only ONE OpenCV wheel may be installed. This project uses opencv-python
# (the Qt overlays never call cv2's own GUI functions, so the headless
# variant buys nothing, and having both installed makes pip pick whichever
# was installed last, which breaks PyInstaller's hook resolution). If
# `pip list` shows opencv-python-headless as well, uninstall it.
opencv-python>=4.9.0

# Pinned below 2.0 because the bundled _rs3vision.pyd was compiled against
# the numpy 1.x C ABI (rust numpy crate 0.22 with numpy 1.26 installed at
# build time). Lift the pin only after rebuilding the .pyd against numpy 2
# and confirming rs3vision.color.find() on a live frame.
numpy>=1.26.0,<2
